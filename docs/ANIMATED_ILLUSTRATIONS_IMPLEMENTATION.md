```typescript
getCurrentTime(): number {
  if (!this.startTime || this.state !== 'playing') return 0;
  const elapsed = performance.now() - this.startTime;
  return Math.min(elapsed, this.audioDuration);
}
```

```typescript
function getIllustrationIcon(elementType: ElementType): string {
  const iconMap: Record<ElementType, string> = {
    box: '◻️', circle: '⭕', arrow: '➡️', text: '📝',
    code: '💻', icon: '🎭', split_panel: '⚡',
    flow_sequence: '🔄', callout: '💬',
  };
  return iconMap[elementType] || '✨';
}

function formatIllustrationLabel(illus: IllustrationSpec): string {
  return `${illus.type} · ${illus.elementType}`;
}
```

**UI Implementation**:

## Technical Decisions

### 1. Word Chunking: 5 Words
**Rationale**: Provides optimal readability without overwhelming the viewer. Research shows 5-7 words is the sweet spot for subtitle comprehension.

### 2. Binary Search for Chunk Lookup
**Rationale**: O(log n) complexity ensures 60fps even with long narrations (100+ chunks).

### 3. RequestAnimationFrame for Time Tracking
**Rationale**: Browser-native timing loop synchronized with GPU refresh for smooth, jank-free updates.

### 4. appearTimingMs for Illustration Control
**Rationale**: Precise millisecond-level control over when each visual element appears, allowing for choreographed sequences.

### 5. Dual Rendering Systems
**Rationale**: Support for both legacy IllustrationSpec and new frame-based VisualScript provides flexibility and backward compatibility.

### 6. CSS Animations over JavaScript
**Rationale**: Hardware-accelerated transforms via CSS provide smooth animations without JavaScript overhead or blocking the main thread.

### 7. Illustration Metadata Reveal
**Rationale**: Shows users what illustrations are coming and when, enhancing anticipation and comprehension. Revealed chips provide positive reinforcement.

## Edge Cases Handled

✅ **Audio without word timestamps** → No subtitles displayed (graceful degradation)
✅ **Scenes without illustrations** → Metadata section not rendered
✅ **Less than 5 words in narration** → Shows all available words in single chunk
✅ **Voice toggle off** → Subtitles hidden automatically
✅ **Scene transition** → Subtitles and illustration timers reset
✅ **Audio playback errors** → Time tracking stops gracefully
✅ **Component unmount** → requestAnimationFrame cleanup prevents memory leaks
✅ **Rapid scene changes** → Subtitle state clears properly
